# Assignment 6: School timetable

**Структура на базата данни**

Базата има пет таблици: classes, subjects, teachers, rooms и lessons. Първите четири
индивидуално отговарят за съхранението на всяка от компонентите на програмата, така че
ако примерно прекръстим учител променяме само една колона. Всяка таблица оперира със
собствено id като PRIMARY KEY и lessons сочи към останалите четири като FOREIGN KEYs,
така че lesson не може да използва несъществуващи teacher или room. За да се отказват
невърможни програми използваме три UNIQUE constraints на lessons: (class, weekday, period),
(teacher, weekday, period) и (room, weekday, period), така че базата сама да спира двойна
информация. Дните се пазят като INT, а не като DATE, защото програмата оперира в постоянен
цикъл от дни от седмицата (1 - 7), което улеснява операциите с информацията.

**Schema**
    CREATE TABLE classes (
        id   INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
        name TEXT NOT NULL UNIQUE
    );

    CREATE TABLE subjects (
        id   INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
        name TEXT NOT NULL UNIQUE
    );

    CREATE TABLE teachers (
        id        INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
        full_name TEXT NOT NULL
    );

    CREATE TABLE rooms (
        id     INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
        number TEXT NOT NULL UNIQUE          -- TEXT: rooms can be '204' or 'Lab-1'
    );

    CREATE TABLE lessons (
        id         INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
        class_id   INTEGER NOT NULL REFERENCES classes(id),
        weekday    SMALLINT NOT NULL CHECK (weekday BETWEEN 1 AND 5),  -- 1=Mon ... 5=Fri
        period     SMALLINT NOT NULL CHECK (period BETWEEN 1 AND 7),
        subject_id INTEGER NOT NULL REFERENCES subjects(id),
        teacher_id INTEGER NOT NULL REFERENCES teachers(id),
        room_id    INTEGER NOT NULL REFERENCES rooms(id),

        CONSTRAINT lessons_class_slot_key   UNIQUE (class_id,   weekday, period),
        CONSTRAINT lessons_teacher_slot_key UNIQUE (teacher_id, weekday, period),
        CONSTRAINT lessons_room_slot_key    UNIQUE (room_id,    weekday, period)
    );

**Constraints check**
    -- 1. Same class, day, period (the one the task requires)
    INSERT INTO lessons (class_id, weekday, period, subject_id, teacher_id, room_id)
    VALUES (1, 1, 1, 4, 2, 3);
    -- ERROR: duplicate key value violates unique constraint "lessons_class_slot_key"
    -- DETAIL: Key (class_id, weekday, period)=(1, 1, 1) already exists.

    -- 2. Teacher in two places: Petrov (2) already teaches 10a on Wed period 1
    INSERT INTO lessons (class_id, weekday, period, subject_id, teacher_id, room_id)
    VALUES (2, 3, 1, 2, 2, 2);
    -- ERROR:  duplicate key value violates unique constraint "lessons_teacher_slot_key"
    -- DETAIL:  Key (teacher_id, weekday, period)=(2, 3, 1) already exists. 

    -- 3. Room double-booked: 204 (1) is already taken Wed period 1
    INSERT INTO lessons (class_id, weekday, period, subject_id, teacher_id, room_id)
    VALUES (2, 3, 1, 2, 3, 1);
    -- ERROR:  duplicate key value violates unique constraint "lessons_room_slot_key"
    -- DETAIL:  Key (room_id, weekday, period)=(1, 3, 1) already exists. 

    -- 4. Wrong foreign key
    INSERT INTO lessons (class_id, weekday, period, subject_id, teacher_id, room_id)
    VALUES (99, 4, 1, 1, 1, 1);
    -- ERROR:  insert or update on table "lessons" violates foreign key constraint "lessons_class_id_fkey"
    -- DETAIL:  Key (class_id)=(99) is not present in table "classes".
